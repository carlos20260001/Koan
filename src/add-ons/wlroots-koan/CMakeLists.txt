# wlroots-koan — Wayland compositor library ported to Koan's app_server.
#
# This CMake build replaces the upstream meson.build. wlroots source files are
# preserved verbatim; only Linux-specific dependencies are adapted through the
# Koan platform layer in backend/koan/ and server/.
#
# NOTE: When included via add_subdirectory() from the Koan build, the parent
# project() call already exists. Do not call project() here. For standalone
# builds, create a wrapper CMakeLists.txt that calls project() first.

cmake_minimum_required(VERSION 3.25)

# wlroots requires the unstable feature macro. Applied per-target (below) so it
# does not leak into the parent Koan build when included via add_subdirectory().

# Dependencies
# ---------------------------------------------------------------------------

find_package(PkgConfig REQUIRED)

# These are mandatory when VOS_WITH_WLROOTS is enabled. Failing here gives a
# clear configure-time error instead of a late "target not found" during
# generation.
pkg_check_modules(WAYLAND_SERVER REQUIRED IMPORTED_TARGET wayland-server)
pkg_check_modules(WAYLAND_CLIENT REQUIRED IMPORTED_TARGET wayland-client)
pkg_check_modules(PIXMAN REQUIRED IMPORTED_TARGET pixman-1)
pkg_check_modules(XKBCOMMON REQUIRED IMPORTED_TARGET xkbcommon)

# libdrm is optional: Koan provides its own display/input stack, so the DRM
# backend, libinput backend, GLES2 and Vulkan renderers are not built. libdrm
# headers are only needed by those backends, which are excluded here.
find_library(LIBM NAMES m)
find_library(LIBRT NAMES rt)

# Protocol XML definitions are vendored under protocol/; the builder does not
# need the wayland-protocols data package to be installed.

# All generated files live in one directory so that the include paths below
# stay simple and version-independent.
set(proto_dir "${CMAKE_CURRENT_BINARY_DIR}/protocol")
set(proto_enum_dir "${proto_dir}/enum/wayland-protocols")
file(MAKE_DIRECTORY "${proto_dir}")
file(MAKE_DIRECTORY "${proto_enum_dir}")

foreach(proto_entry IN LISTS WLR_PROTOCOL_ENTRIES)
    string(REPLACE "=" ";" proto_parts "${proto_entry}")
    list(GET proto_parts 0 proto_name)
    list(GET proto_parts 1 proto_path)
    get_filename_component(proto_basename "${proto_path}" NAME)
    # Strip the .xml extension to match wlroots' expected include names
    string(REGEX REPLACE "\\.xml$" "" proto_name_stripped "${proto_basename}")

    # Private protocol C implementation
    set(proto_c "${proto_dir}/${proto_name_stripped}-protocol.c")
    add_custom_command(
        OUTPUT "${proto_c}"
        COMMAND ${WAYLAND_SCANNER} private-code "${proto_path}" "${proto_c}"
        DEPENDS "${proto_path}"
        COMMENT "Generating wayland protocol ${proto_name}"
        VERBATIM
    )
    list(APPEND WLR_PROTOCOL_SRCS "${proto_c}")

    # Server protocol header
    set(proto_server_h "${proto_dir}/${proto_name_stripped}-protocol.h")
    add_custom_command(
        OUTPUT "${proto_server_h}"
        COMMAND ${WAYLAND_SCANNER} server-header "${proto_path}" "${proto_server_h}"
        DEPENDS "${proto_path}"
        COMMENT "Generating wayland server header ${proto_name}"
        VERBATIM
    )
    list(APPEND WLR_PROTOCOL_SERVER_HDRS "${proto_server_h}")

    # Public wlroots headers include <wayland-protocols/NAME-enum.h>.
    # wayland-scanner 1.23.1's enum-header mode emits the enums without
    # validators, and the validator definitions live inside the same
    # NAME_ENUM include guard as the enum in the server header. Including
    # such an enum-only header before the server header would therefore
    # suppress the validators. Forward to the server header instead: it
    # provides both enums and validators from the same XML, and works with
    # every scanner version. The forwarder is a configure-time file, but it
    # is also registered as a generated target input so the build graph has
    # an explicit dependency on it.
    set(proto_enum_h "${proto_enum_dir}/${proto_name_stripped}-enum.h")
    file(WRITE "${proto_enum_h}"
        "#include \"${proto_name_stripped}-protocol.h\"\n")
    set_source_files_properties("${proto_enum_h}" PROPERTIES GENERATED TRUE)
    list(APPEND WLR_PROTOCOL_SERVER_HDRS "${proto_enum_h}")
endforeach()

# ---------------------------------------------------------------------------
# wl_fixes backport for wayland-server < 1.24
# ---------------------------------------------------------------------------
# wlroots 0.20.2 references the core wl_fixes interface, which was added in
# wayland 1.24. Generate the version 1 interface locally when the builder's
# wayland is older; wlroots-compat.h includes the generated header and
# wlroots-compat.c provides the few missing core APIs.
if(WAYLAND_SERVER_VERSION VERSION_LESS 1.24)
    set(WLR_FIXES_XML "${CMAKE_CURRENT_SOURCE_DIR}/compat/fixes.xml")
    set(WLR_FIXES_C "${proto_dir}/fixes-protocol.c")
    set(WLR_FIXES_H "${proto_dir}/fixes-protocol.h")
    add_custom_command(
        OUTPUT "${WLR_FIXES_C}"
        COMMAND ${WAYLAND_SCANNER} private-code "${WLR_FIXES_XML}" "${WLR_FIXES_C}"
        DEPENDS "${WLR_FIXES_XML}"
        COMMENT "Generating wl_fixes compatibility protocol"
        VERBATIM
    )
    add_custom_command(
        OUTPUT "${WLR_FIXES_H}"
        COMMAND ${WAYLAND_SCANNER} server-header "${WLR_FIXES_XML}" "${WLR_FIXES_H}"
        DEPENDS "${WLR_FIXES_XML}"
        COMMENT "Generating wl_fixes compatibility header"
        VERBATIM
    )
    list(APPEND WLR_PROTOCOL_SRCS "${WLR_FIXES_C}")
    list(APPEND WLR_PROTOCOL_SERVER_HDRS "${WLR_FIXES_H}")
    message(STATUS "wlroots-koan: wayland-server ${WAYLAND_SERVER_VERSION} "
        "lacks wl_fixes; building compatibility interface")
endif()

target_include_directories(wlroots PUBLIC
    # Single header root. Do not add include/wlr as well: that makes
    # "types/foo.h" resolve to the public enum-only header instead of the
    # private shim in include/types/.
    "${CMAKE_CURRENT_SOURCE_DIR}/include"
    "${CMAKE_CURRENT_BINARY_DIR}/include"
    # Generated protocol sources and headers.
    "${CMAKE_CURRENT_BINARY_DIR}/protocol"
    # Generated <wayland-protocols/NAME-enum.h> forwarders. This directory
    # must precede the static fallback copy below.
    "${CMAKE_CURRENT_BINARY_DIR}/protocol/enum"
    "${CMAKE_CURRENT_SOURCE_DIR}/server"
    "${CMAKE_CURRENT_SOURCE_DIR}/compat"
    # Fallback for enum headers of protocols that are not generated here.
    "${CMAKE_CURRENT_SOURCE_DIR}/protocol/enum"
)

target_compile_options(wlroots PRIVATE -include wlroots-compat.h)

target_include_directories(wlroots PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}"
    "${CMAKE_CURRENT_SOURCE_DIR}/backend/koan"
)

target_compile_definitions(wlroots PUBLIC
    WLR_USE_UNSTABLE
    WLR_PRIVATE=
)
target_compile_definitions(wlroots PRIVATE
    _POSIX_C_SOURCE=200809L
    # The compat header is force-included (-include) into every source and
    # pulls in glibc headers, so <features.h> is processed before xcursor.c
    # can define _DEFAULT_SOURCE itself. Define it up front so DT_* remains
    # visible despite _POSIX_C_SOURCE.
    _DEFAULT_SOURCE
)

target_link_libraries(wlroots PUBLIC
    PkgConfig::WAYLAND_SERVER
    PkgConfig::WAYLAND_CLIENT
    PkgConfig::PIXMAN
    PkgConfig::XKBCOMMON
    ${LIBM}
    ${LIBRT}
)

# The library is linked into app_server, which may be built as PIE.
set_target_properties(wlroots PROPERTIES
    POSITION_INDEPENDENT_CODE ON
    VERSION 0.20.2
    SOVERSION 0.20
)
