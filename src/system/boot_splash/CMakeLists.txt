# vos-splash: the Haiku boot splash (boot loader video_splash.cpp + kernel
# boot_splash.cpp) on a DRM dumb buffer or fbdev. It runs from the initramfs,
# so it links statically against libc and zlib only: no libbe, no libroot.

# Trademarks: only the air/OS artwork may be compiled in. The Haiku logo
# headers (images-tm*.h, images-sans-tm.h, and images.h, which selects them
# for official builds) must never reach this target.
set(VOS_SPLASH_IMAGES
	"${PROJECT_SOURCE_DIR}/headers/private/kernel/boot/images-airos.h"
	CACHE FILEPATH "Boot splash image header made by generate_boot_screen")
get_filename_component(_vos_splash_images_name "${VOS_SPLASH_IMAGES}" NAME)
if(_vos_splash_images_name MATCHES "^images(-tm.*|-sans-tm.*)?\\.h$")
	message(FATAL_ERROR
		"VOS_SPLASH_IMAGES=${VOS_SPLASH_IMAGES}: the Haiku boot splash images "
		"are Haiku trademarks and must not ship. Use "
		"headers/private/kernel/boot/images-airos.h or other artwork you may "
		"distribute.")
endif()
if(NOT EXISTS "${VOS_SPLASH_IMAGES}")
	message(FATAL_ERROR "VOS_SPLASH_IMAGES=${VOS_SPLASH_IMAGES} does not exist")
endif()

file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/splash_images.h" CONTENT
"// Generated by src/system/boot_splash/CMakeLists.txt: the splash artwork.
#if defined(HAIKU_DISTRO_COMPATIBILITY_OFFICIAL)
#	error \"vos-splash must not be built with the Haiku trademark images\"
#endif
#include \"@VOS_SPLASH_IMAGES@\"
" @ONLY)

find_package(ZLIB REQUIRED)
find_library(VOS_SPLASH_ZLIB_STATIC NAMES libz.a)
if(NOT VOS_SPLASH_ZLIB_STATIC)
	message(FATAL_ERROR "vos-splash links zlib statically: install libz.a (zlib1g-dev)")
endif()

# The drawing code shared by the program and its tests.
add_library(vos_splash_core STATIC
	video_splash.cpp
	boot_splash.cpp
	splash_stages.cpp
)
target_include_directories(vos_splash_core PUBLIC
	"${CMAKE_CURRENT_SOURCE_DIR}"
	"${CMAKE_CURRENT_BINARY_DIR}"
	"${PROJECT_SOURCE_DIR}/headers/private/kernel"
)
target_include_directories(vos_splash_core PRIVATE ${ZLIB_INCLUDE_DIRS})
# Same prelude as every other target here (SupportDefs.h types, __VOS__);
# it is header-only, so the program still needs no Haiku library.
target_compile_options(vos_splash_core PUBLIC
	-include LinuxBuildCompatibility.h -fno-exceptions -fno-rtti)

add_executable(vos-splash
	splash_display.cpp
	vos_splash.cpp
)
target_link_libraries(vos-splash PRIVATE vos_splash_core
	"${VOS_SPLASH_ZLIB_STATIC}")
# Static: the initramfs hook copies one file and needs no library closure.
target_link_options(vos-splash PRIVATE -static -no-pie)
set_target_properties(vos-splash PROPERTIES POSITION_INDEPENDENT_CODE OFF)
